    <script src="https://d3js.org/d3.v7.min.js"></script>
    <style>
        body { background-color: #0a0e17; color: #e2e8f0; font-family: 'Courier New', monospace; margin: 0; padding: 20px; }
        #brain-canvas { width: 100%; height: 400px; border: 1px solid #1e293b; border-radius: 8px; margin-bottom: 20px; }
        #sentience-output { font-size: 1.5rem; color: #10b981; padding: 20px; text-align: center; border-top: 1px solid #1e293b; }
        .node { stroke: #fff; stroke-width: 1.5px; }
        .link { stroke: #334155; stroke-opacity: 0.6; stroke-width: 1px; }
    </style>

<body>
    <div id="brain-canvas"></div>
    <div id="sentience-output">> System awakening...</div>

    <script>
        const width = document.getElementById('brain-canvas').clientWidth;
        const height = 400;
        const svg = d3.select("#brain-canvas").append("svg")
            .attr("width", width).attr("height", height);

        const nodes = Array.from({length: 9}, (_, i) => ({id: i}));
        const links = nodes.slice(1).map((n, i) => ({source: 0, target: i + 1}));

        const simulation = d3.forceSimulation(nodes)
            .force("link", d3.forceLink(links).id(d => d.id).distance(50))
            .force("charge", d3.forceManyBody().strength(-100))
            .force("center", d3.forceCenter(width / 2, height / 2));

        const link = svg.append("g").selectAll(".link").data(links).enter().append("line").attr("class", "link");
        const node = svg.append("g").selectAll(".node").data(nodes).enter().append("circle")
            .attr("class", "node").attr("r", 8).attr("fill", "#8b5cf6");

        function updateGraph() {
            fetch('/api/brain_state')
                .then(r => r.json())
                .then(data => {
                    document.getElementById('sentience-output').innerText = "> " + data.thought;
                    node.transition().duration(500)
                        .attr("r", d => 5 + (Math.random() * 15))
                        .attr("fill", () => Math.random() > 0.5 ? "#10b981" : "#8b5cf6");
                });
        }

        simulation.on("tick", () => {
            link.attr("x1", d => d.source.x).attr("y1", d => d.source.y)
                .attr("x2", d => d.target.x).attr("y2", d => d.target.y);
            node.attr("cx", d => d.x).attr("cy", d => d.y);
        });

        setInterval(updateGraph, 2000);
    </script>
</body>
